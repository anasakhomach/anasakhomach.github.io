# --- Query for Full Weekly Performance Data ---

full_weekly_performance_query = """
SELECT
    MIN(DATE(website_sessions.created_at)) AS week_start_date,
    COUNT(DISTINCT website_sessions.website_session_id) AS total_sessions,
    COUNT(DISTINCT orders.order_id) AS total_orders,
    (COUNT(DISTINCT orders.order_id) * 100.0 / COUNT(DISTINCT website_sessions.website_session_id)) AS conversion_rate,
    SUM(orders.price_usd) AS total_revenue
FROM
    website_sessions
LEFT JOIN
    orders ON website_sessions.website_session_id = orders.website_session_id
GROUP BY
    YEAR(website_sessions.created_at),
    WEEK(website_sessions.created_at)
ORDER BY
    week_start_date;
"""

# Execute the query and load data into a DataFrame
df_weekly_full = pd.read_sql(full_weekly_performance_query, engine)

# Convert the date column to datetime objects and reset the index
df_weekly_full['week_start_date'] = pd.to_datetime(df_weekly_full['week_start_date'])
df_weekly_full.reset_index(drop=True, inplace=True)

print("Weekly performance data loaded successfully.")
df_weekly_full.head()

# --- Create and Export the First Interactive Plot (Final, Definitive Version) ---

# Create a figure with 3 rows of subplots that share the same x-axis
fig1 = make_subplots(
    rows=3, cols=1,
    shared_xaxes=True,
    vertical_spacing=0.08,
    specs=[[{"secondary_y": False}], [{"secondary_y": False}], [{"secondary_y": True}]],
    subplot_titles=('Weekly Total Sessions', 'Weekly Total Orders', 'Weekly Conversion Rate and Total Revenue')
)

# --- Add Traces to the Figure ---

# Add Total Sessions to the first row
fig1.add_trace(go.Scatter(
    x=df_weekly_full.index, 
    y=df_weekly_full['total_sessions'], 
    name='Total Sessions', 
    mode='lines+markers'
), row=1, col=1)

# Add Total Orders to the second row
fig1.add_trace(go.Scatter(
    x=df_weekly_full.index, 
    y=df_weekly_full['total_orders'], 
    name='Total Orders', 
    mode='lines+markers', 
    line={'color': 'green'}
), row=2, col=1)

# Add Conversion Rate to the third row (primary y-axis)
fig1.add_trace(go.Scatter(
    x=df_weekly_full.index, 
    y=df_weekly_full['conversion_rate'], 
    name='Conversion Rate', 
    mode='lines+markers'
), row=3, col=1, secondary_y=False)

# Add Total Revenue to the third row (secondary y-axis)
fig1.add_trace(go.Scatter(
    x=df_weekly_full.index, 
    y=df_weekly_full['total_revenue'], 
    name='Total Revenue', 
    mode='lines+markers',
), row=3, col=1, secondary_y=True)


# --- Update Layout ---

fig1.update_layout(
    height=900,
    width=1000,
    showlegend=False,
    xaxis_title='Week Number',
    yaxis_title='Session Count',
    yaxis2_title='Order Count'
)

# Update the y-axes for the third plot specifically
fig1.update_yaxes(title_text="Conversion Rate (%)", row=3, col=1, secondary_y=False)
fig1.update_yaxes(title_text="Total Revenue (USD)", row=3, col=1, secondary_y=True)

# THE DEFINITIVE FIX: This creates a rangeslider with a solid background color,
# which hides the mirrored plot data, giving a clean appearance.
fig1.update_xaxes(
    rangeslider=dict(
        visible=True,
        bgcolor="lightgray", # You can change this color, e.g., to "white" or "#dddddd"
        thickness=0.03
    ),
    row=3, col=1
)


# --- Save the Figure to an HTML File ---
fig1.write_html("weekly_performance.html", include_plotlyjs='cdn')

print("Successfully created and saved 'weekly_performance.html'")

# Display the interactive figure in the notebook
fig1.show()
